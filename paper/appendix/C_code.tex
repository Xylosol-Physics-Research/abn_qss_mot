│   ├── benchmark.py        # Single-frame benchmark

│   ├── test_nullspace.py
│   └── test_solver.py
└── results/
    ├── single_frame.txt
    ├── multi_frame.txt
    └── statistical.txt
\end{verbatim}

\subsection{Reproduction Commands}

\begin{verbatim}
# Install dependencies
pip install -e .

# Run all tests
pytest tests/ -v --cov=core

# Single-frame benchmark
python -m experiments.benchmark

# Multi-frame benchmark
python -m experiments.benchmark_multiframe

# Cooperative benchmark
python -m experiments.cooperative_benchmark

# Statistical significance
python -m experiments.statistical_significance

# Regenerate all figures
python -m experiments.plotting.plot_all
\end{verbatim}
